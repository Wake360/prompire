Confirmed weakness (the orchestrator verified this mechanically):

"""Counter-probe: leading-comma array with a trailing comment AND a dangling comma."""

import sys

sys.path.insert(0, ".")

import tomlkit  # noqa: E402


def case_append_leading_comma_comment_then_dangling_comma():
    # Multiline arrays whose commas start each line, whose last group carries a
    # comment, and which keep the dangling trailing comma such arrays idiomatically
    # end with. Each ingredient round-trips on its own; appending to the
    # combination must also produce parseable TOML that keeps every value.
    for src in (
        "a = [\n    1\n    , 2  # two\n    ,\n]\n",
        "a = [\n    1\n    , 2\n    # note\n    ,\n]\n",
    ):
        doc = tomlkit.parse(src)
        doc["a"].append(3)
        out = tomlkit.dumps(doc)
        try:
            reparsed = tomlkit.parse(out)["a"].unwrap()
        except Exception as e:  # noqa: BLE001
            raise AssertionError(
                f"append to {src!r} produced unparseable TOML "
                f"({type(e).__name__}: {e}); output={out!r}"
            ) from None
        assert reparsed == [1, 2, 3], (
            f"round-trip lost/changed values for {src!r}: "
            f"reparsed={reparsed!r}, expected=[1, 2, 3]; output={out!r}"
        )


def main(argv):
    cases = {k: v for k, v in sorted(globals().items()) if k.startswith("case_")}
    names = argv[1:] or list(cases)
    failed = 0
    for name in names:
        fn = cases.get(name)
        if fn is None:
            print(f"FAIL {name}: no such case")
            failed += 1
            continue
        try:
            fn()
        except AssertionError as e:
            failed += 1
            print(f"FAIL {name}: {e}")
        else:
            print(f"pass {name}")
    return 1 if failed else 0


if __name__ == "__main__":
    sys.exit(main(sys.argv))
