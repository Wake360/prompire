  import sys

  sys.path.insert(0, ".")

  from tabulate import tabulate  # noqa: E402

  TABLE = [["spam", 41.9999], ["eggs", "451.0"]]
  HEADERS = ["strings", "numbers"]


  def _lines(*a, **kw):
      return tabulate(*a, **kw).split("\n")


  def _assert_no_trailing_ws(lines, label):
      bad = [(i, ln) for i, ln in enumerate(lines) if ln != ln.rstrip()]
      assert not bad, f"{label}: trailing whitespace on lines {bad!r}"


  def case_data_rows_no_trailing_whitespace():
      lines = _lines(TABLE, HEADERS, tablefmt="asciidoc")
      _assert_no_trailing_ws(lines[3:5], "data rows")


  def case_header_row_no_trailing_whitespace():
      lines = _lines(TABLE, HEADERS, tablefmt="asciidoc")
      _assert_no_trailing_ws([lines[2]], "header row")


  def case_headerless_rows_no_trailing_whitespace():
      lines = _lines(TABLE, tablefmt="asciidoc")
      _assert_no_trailing_ws(lines, "headerless table")


  def case_exact_output_with_headers():
      expected = "\n".join(
          [
              '[cols="<11,>11",options="header"]',
              "|====",
              "| strings   |   numbers",
              "| spam      |   41.9999",
              "| eggs      |  451",
              "|====",
          ]
      )
      got = tabulate(TABLE, HEADERS, tablefmt="asciidoc")
      assert got == expected, f"with headers: {got!r}"


  def case_exact_output_headerless():
      expected = "\n".join(
          [
              '[cols="<6,>10"]',
              "|====",
              "| spam |  41.9999",
              "| eggs | 451",
              "|====",
          ]
      )
      got = tabulate(TABLE, tablefmt="asciidoc")
      assert got == expected, f"headerless: {got!r}"


  def case_right_aligned_last_column_padding_removed():

      assert lines[4] == "| eggs      |  451", f"decimal-aligned row: {lines[4]!r}"

      assert lines[3] == "| x   |", f"row with empty last col: {lines[3]!r}"
      assert lines[4] == "| yy  |", f"row with empty last col: {lines[4]!r}"


  def case_single_empty_column_row():
      lines = _lines([["a"], [""]], tablefmt="asciidoc")
      assert lines[2] == "| a", f"single-col row: {lines[2]!r}"

notes: >-
  The defect is one line: asciidoc is the only format whose callable datarow
  joins pre-padded cells without the .rstrip() that _build_simple_row and the
  mediawiki/html row builders apply, so the last cell's padding leaks to
  end-of-line on header and data rows alike. The spec is deliberately wider than
  "data rows": it covers the header row (same expression), all alignments (right
  and center leak more than one space), and pins the boundary that stripping must
  not swallow '|' separators when the trailing cell is empty. Fixing it breaks
  three fixtures that encode the old bytes — test_asciidoc, test_asciidoc_headerless,
  and the README asciidoc doctest — all three are in scope.
