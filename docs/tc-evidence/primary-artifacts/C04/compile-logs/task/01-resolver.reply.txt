  import sys

  sys.path.insert(0, '.')

  from boltons.setutils import IndexedSet


  def _with_dead_slots():
      """An IndexedSet retaining uncompacted dead_indices."""
      s = IndexedSet(range(1000))
      s.pop(10)
      s.pop(20)
      assert s.dead_indices, 'fixture failed to retain dead indices'
      return s


  def _expect_index_error(fn, what):
      try:
          val = fn()
      except IndexError as ie:
          return ie
      raise AssertionError(f'{what}: expected IndexError, got {val!r}')


  def case_getitem_too_negative_raises():
      s = IndexedSet(range(4))
      for i in (-5, -6, -8, -9, -100):
          _expect_index_error(lambda: s[i], f'IndexedSet(range(4))[{i}]')


  def case_getitem_too_negative_with_dead_slots_raises():
      s = _with_dead_slots()
      n = len(s)
      for i in (-n - 1, -n - 5, -2 * n, -2 * n - 3):
          _expect_index_error(lambda: s[i], f'dead-slot set[{i}]')


  def case_getitem_negative_out_of_range_message():
      s = IndexedSet(range(4))
      pos = _expect_index_error(lambda: s[4], 'set[4]')
      neg = _expect_index_error(lambda: s[-5], 'set[-5]')
      if str(neg) != str(pos):
          raise AssertionError(
              f'negative out-of-range message {str(neg)!r} differs from '
              f'positive out-of-range message {str(pos)!r}')


  def case_pop_too_negative_raises_and_leaves_set_intact():
      for i in (-5, -6, -9, -100):
          s = IndexedSet(range(4))
          before = list(s)
          _expect_index_error(lambda: s.pop(i), f'IndexedSet(range(4)).pop({i})')
          if list(s) != before:
              raise AssertionError(f'pop({i}) mutated the set: {list(s)!r}')
