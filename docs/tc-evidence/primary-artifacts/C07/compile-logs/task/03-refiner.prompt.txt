import sys

sys.path.insert(0, '.')

from decimal import Decimal

import parse


def case_precision_zero_search_prefers_full_number():
    """Widening .0f/.0F must only widen: an unanchored match must not stop at
    the integer prefix of a number that does have a fractional part."""
    r = parse.search("{:.0f}", "1.5")
    assert r is not None, "search('{:.0f}', '1.5') found nothing"
    assert r.fixed[0] == 1.5, "search('{:.0f}', '1.5') gave %r" % (r.fixed[0],)

    r = parse.search("value = {:.0f}", "value = -2.25 end")
    assert r is not None, "search with literal prefix found nothing"
    assert r.fixed[0] == -2.25, "search('value = {:.0f}') gave %r" % (r.fixed[0],)

    r = parse.search("{:.0F}", "7.25")
    assert r is not None, "search('{:.0F}', '7.25') found nothing"
    assert r.fixed[0] == Decimal("7.25"), "search('{:.0F}') gave %r" % (r.fixed[0],)

    got = [m.fixed[0] for m in parse.findall("{:.0f}", "1.5 2.75")]
    assert got == [1.5, 2.75], "findall('{:.0f}', '1.5 2.75') gave %r" % (got,)
