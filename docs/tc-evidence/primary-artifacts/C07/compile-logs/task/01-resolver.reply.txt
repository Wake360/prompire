  import sys

  sys.path.insert(0, '.')

  from decimal import Decimal

  import parse


  def _fixed(fmt, string):
      r = parse.parse(fmt, string)
      assert r is not None, "parse(%r, %r) returned None" % (fmt, string)
      return r.fixed


  def case_f_precision_zero_matches_integer():
      for text, want in [("42", 42.0), ("0", 0.0), ("1234567", 1234567.0)]:
          got = _fixed("{:.0f}", text)[0]
          assert got == want, "{:.0f} on %r gave %r" % (text, got)
          assert isinstance(got, float), "{:.0f} on %r gave %r" % (text, type(got))
      got = _fixed("Count = {:.0f}", "Count = 42")[0]
      assert got == 42.0, "literal-surrounded field gave %r" % (got,)


  def case_f_precision_zero_signs():
      for text, want in [("-42", -42.0), ("+42", 42.0), (" 42", 42.0), ("-0", 0.0)]:
          got = _fixed("{:.0f}", text)[0]
          assert got == want, "{:.0f} on %r gave %r" % (text, got)


  def case_f_precision_zero_group_indexes_intact():
      got = _fixed("{:.0f} {:.0f} {:d}", "1 2 3")
      assert got == (1.0, 2.0, 3), "fixed fields gave %r" % (got,)
      r = parse.parse("{a:.0f} {b:.0f} {c:d}", "-1 2 3")
      assert r is not None, "named precision-0 float fields did not match"
      assert r.named == {"a": -1.0, "b": 2.0, "c": 3}, "named gave %r" % (r.named,)


  def case_F_precision_zero_matches_integer():
      for text, want in [("7", Decimal("7")), ("0", Decimal("0"))]:
          got = _fixed("{:.0F}", text)[0]
          assert got == want, "{:.0F} on %r gave %r" % (text, got)
          assert isinstance(got, Decimal), "{:.0F} on %r gave %r" % (text, type(got))
      got = _fixed("{:.0F} {:d}", "7 3")
      assert got == (Decimal("7"), 3), "{:.0F} {:d} gave %r" % (got,)


  def case_e_precision_zero_matches_integer_mantissa():
      for text, want in [("1e+00", 1.0), ("-1e+00", -1.0), ("+2E5", 200000.0)]:
          got = _fixed("{:.0e}", text)[0]
          assert got == want, "{:.0e} on %r gave %r" % (text, got)
      got = _fixed("{:.0e} {:d}", "1e+00 5")
      assert got == (1.0, 5), "{:.0e} {:d} gave %r" % (got,)


  def case_precision_zero_with_width():
      got = _fixed("{:8.0f}", "       3")[0]
      assert got == 3.0, "{:8.0f} on padded input gave %r" % (got,)
      got = _fixed("{:08.0f}", "00000003")[0]
      assert got == 3.0, "{:08.0f} on zero-padded input gave %r" % (got,)


  def case_precision_zero_format_roundtrip():
      for value in [1.0, -1.0, 0.0, 12345.0]:
          for spec in [".0f", ".0e"]:
              text = format(value, spec)
              want = float(text)
              got = _fixed("{:%s}" % spec, text)[0]
              assert got == want, "roundtrip %r of %r via %r gave %r" % (
                  spec, value, text, got,
              )
      text = format(Decimal("7"), ".0f")
      got = _fixed("{:.0F}", text)[0]
      assert got == Decimal("7"), "roundtrip .0F via %r gave %r" % (text, got)


  def case_precision_zero_still_matches_fractional():
      for fmt, text, want in [
          ("{:.0f}", "1.0", 1.0),
          ("{:.0f}", ".5", 0.5),
          ("{:.0f}", "-2.25", -2.25),
          ("{:.0e}", "1.0e+00", 1.0),
      ]:
          got = _fixed(fmt, text)[0]
          assert got == want, "%s on %r gave %r" % (fmt, text, got)
      got = _fixed("{:.0F}", "7.0")[0]
