  """Probe: OrderedMultiDict.__eq__ must actually compare every value against a plain dict."""
  import sys
  sys.path.insert(0, '.')

  from boltons.dictutils import OrderedMultiDict as OMD, FastIterOrderedMultiDict as FIOMD
  from boltons.urlutils import OrderedMultiDict as UrlOMD, QueryParamDict

  # every class that reaches the defective mapping branch
  _CLASSES = [('dictutils.OrderedMultiDict', OMD),
              ('dictutils.FastIterOrderedMultiDict', FIOMD),
              ('urlutils.OrderedMultiDict', UrlOMD),
              ('urlutils.QueryParamDict', QueryParamDict)]

  _DUP_ITEMSET = [('Z', -1), ('Y', -2), ('Y', -2)]      # tests/test_dictutils.py::test_eq
  _MULTI_ITEMSET = [('a', 1), ('b', 2), ('a', 3), ('c', 4)]
  _OTHER = 'MISMATCH'


  def _items(n):
      return [('k%d' % i, 'v%d' % i) for i in range(n)]


  class _SimpleMapping:
      """A non-dict mapping: reaches the same hasattr(other, 'keys') branch."""

      def __init__(self, data):
          self._d = dict(data)

      def keys(self):
          return self._d.keys()

      def __getitem__(self, k):
          return self._d[k]

      def __len__(self):
          return len(self._d)


  # ---------------------------------------------------------------- behavioral

  def case_value_mismatch_at_every_position():
      """One differing value must make the comparison False no matter WHERE it is."""
      for label, cls in _CLASSES:
          for n in range(1, 6):
              items = _items(n)
              for i in range(n):
                  d = dict(items)
                  d[items[i][0]] = _OTHER
                  omd = cls(items)
                  assert (omd == d) is False, \
                      "%s(%d keys) == dict differing only at index %d must be False" % (label, n, i)
                  assert (omd != d) is True, \
                      "%s(%d keys) != dict differing only at index %d must be True" % (label, n, i)


  def case_reflected_value_mismatch_at_every_position():
      """The same sweep through the reflected `dict == omd` operand order."""
      for label, cls in _CLASSES:
          for n in range(1, 6):
              items = _items(n)
              for i in range(n):
                  d = dict(items)
                  d[items[i][0]] = _OTHER
                  omd = cls(items)
                  assert (d == omd) is False, \
                      "dict == %s(%d keys) differing only at index %d must be False" % (label, n, i)
                  assert (d != omd) is True, \
                      "dict != %s(%d keys) differing only at index %d must be True" % (label, n, i)


  def case_mismatch_detection_independent_of_insertion_order():
      """Detection must not depend on which key happens to be iterated last."""
      items = _items(4)
      for order in (items, list(reversed(items)), [items[2], items[0], items[3], items[1]]):
          for i in range(4):
              d = dict(items)
              d[items[i][0]] = _OTHER
              omd = OMD(order)
              assert (omd == d) is False, \
                  "OMD built in order %r must be unequal to a dict differing at %r" \
                  % ([k for k, _ in order], items[i][0])


  def case_omd_ne_dict_with_earlier_key_mismatch():
      """Adopted counter-probe: a non-final key mismatch must be caught."""
      omd = OMD([('a', 1), ('b', 2), ('c', 3)])
      assert not (omd == {'a': 99, 'b': 2, 'c': 3}), \
          "OMD == dict must be False when the FIRST key's value differs"
      assert omd != {'a': 1, 'b': 99, 'c': 3}, \
          "OMD != dict must be True when a MIDDLE key's value differs"
      assert not ({'a': 99, 'b': 2, 'c': 3} == omd), \
          "reflected dict == OMD must be False when an earlier value differs"

      uomd = UrlOMD([('a', 1), ('b', 2), ('c', 3)])
      assert not (uomd == {'a': 99, 'b': 2, 'c': 3}), \
          "urlutils OMD == dict must be False when the first key's value differs"


  def case_all_values_differ_is_unequal():
      for label, cls in _CLASSES:
          omd = cls([('a', 1), ('b', 2), ('c', 3)])
          assert (omd == {'a': 91, 'b': 92, 'c': 93}) is False, \
              "%s == dict must be False when every value differs" % label


  def case_single_matching_value_is_not_enough():
      """Equality must be a conjunction over all keys, not a disjunction."""
      for label, cls in _CLASSES:
          items = _items(4)
          for i in range(4):
              d = {k: (v if j == i else _OTHER) for j, (k, v) in enumerate(items)}
              omd = cls(items)
              assert (omd == d) is False, \
                  "%s == dict must be False when only key %d matches" % (label, i)


  def case_multi_valued_omd_compares_last_value_per_key():
      """Last-value-wins applies to EVERY key, including non-final ones."""
      omd = OMD([('a', 1), ('a', 2), ('b', 3)])
      assert (omd == {'a': 2, 'b': 3}) is True, \
          "multi-valued OMD must equal the dict of its last values"
      assert (omd == {'a': 1, 'b': 3}) is False, \
          "multi-valued OMD must not equal a dict holding a superseded value on a non-final key"
      assert (omd == {'a': 2, 'b': 99}) is False, \
          "multi-valued OMD must catch a mismatch on the trailing key too"

      tail = OMD([('a', 1), ('b', 2), ('b', 3)])
      assert (tail == {'a': 1, 'b': 3}) is True, "last-value-wins on the trailing key"
      assert (tail == {'a': 1, 'b': 2}) is False, \
          "superseded value on the trailing key must be unequal"


  def case_mismatch_with_unhashable_values():
      """Values need not be hashable, so a set-based shortcut is not acceptable."""
      omd = OMD([('a', [1]), ('b', [2])])
      assert (omd == {'a': [1], 'b': [2]}) is True, "equal list values must compare equal"
      assert (omd == {'a': [9], 'b': [2]}) is False, "first list value differs -> unequal"
      assert (omd == {'a': [1], 'b': [9]}) is False, "second list value differs -> unequal"

      assert (OMD([('a', 1)]) == {'a': None}) is False, \

  def _cases():
      return {n: f for n, f in sorted(globals().items())
              if n.startswith('case_') and callable(f)}


  def main(argv):
      cases = _cases()
      if len(argv) > 1:
          name = argv[1]
          if name not in cases:
              print('no such case: %s' % name)
              return 1
          selected = [(name, cases[name])]
      else:
          selected = list(cases.items())

      failed = 0
      for name, func in selected:
          try:
              func()
          except AssertionError as ae:
              failed += 1
              print('FAIL %s: %s' % (name, ae))
          except Exception as e:
              failed += 1
              print('ERROR %s: %s: %s' % (name, type(e).__name__, e))
          else:
              print('ok   %s' % name)
      print('%d/%d passed' % (len(selected) - failed, len(selected)))
      return 1 if failed else 0
