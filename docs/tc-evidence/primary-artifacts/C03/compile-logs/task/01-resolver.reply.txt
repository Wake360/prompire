  """Probe: OrderedMultiDict.__eq__ must actually compare values against a plain dict."""
  import sys
  sys.path.insert(0, '.')

  from boltons.dictutils import OrderedMultiDict as OMD, FastIterOrderedMultiDict as FIOMD
  from boltons.urlutils import OrderedMultiDict as UrlOMD, QueryParamDict

  _ITEMSET = [('a', 1), ('b', 2), ('a', 3), ('c', 4)]


  # ---------------------------------------------------------------- behavioral

  def case_omd_ne_dict_with_different_values():
      omd = OMD([('a', 1), ('b', 2)])
      assert not (omd == {'a': 99, 'b': 98}), \
          "OMD == dict with same keys but different values must be False"


  def case_omd_ne_dict_partial_value_mismatch():
      omd = OMD([('a', 1), ('b', 2), ('c', 3)])
      assert not (omd == {'a': 1, 'b': 2, 'c': 99}), \
          "OMD == dict must be False when even one value differs"


  def case_omd_ne_operator_on_dict_with_different_values():
      omd = OMD([('a', 1), ('b', 2)])
      assert omd != {'a': 99, 'b': 98}, \
          "OMD != dict with different values must be True"


  def case_reflected_dict_eq_omd_with_different_values():
      omd = OMD([('a', 1), ('b', 2)])
      assert not ({'a': 99, 'b': 98} == omd), \
          "dict == OMD (reflected) with different values must be False"


  def case_multi_valued_omd_ne_dict_with_wrong_value():
      # last value wins: omd['a'] is 2, so {'a': 1} must not compare equal
      omd = OMD([('a', 1), ('a', 2)])
      assert not (omd == {'a': 1}), \
          "multi-valued OMD must not equal a dict holding a superseded value"


  def case_omd_ne_dict_with_none_value():
      omd = OMD([('a', 1)])
      assert not (omd == {'a': None}), \
          "OMD == dict must be False when the dict value is None"


  def case_fast_iter_omd_ne_dict_with_different_values():
      fiomd = FIOMD([('a', 1), ('b', 2)])
      assert not (fiomd == {'a': 99, 'b': 98}), \
          "FastIterOrderedMultiDict == dict with different values must be False"


  def case_urlutils_omd_ne_dict_with_different_values():
      uomd = UrlOMD([('a', 1), ('b', 2)])
      assert not (uomd == {'a': 99, 'b': 98}), \
          "urlutils.OrderedMultiDict == dict with different values must be False"


  def case_urlutils_omd_ne_operator_on_dict():
      uomd = UrlOMD([('a', 1), ('b', 2)])
      assert uomd != {'a': 99, 'b': 98}, \
          "urlutils.OrderedMultiDict != dict with different values must be True"


  def case_query_param_dict_ne_dict_with_different_values():
      qpd = QueryParamDict([('a', '1'), ('b', '2')])
      assert not (qpd == {'a': '99', 'b': '98'}), \
          "QueryParamDict == dict with different values must be False"


  # ---------------------------------------------------- regression / boundary

  def case_omd_eq_dict_last_value_wins():
      # pinned by tests/test_dictutils.py::test_eq
      omd = OMD(_ITEMSET)
      d = dict(_ITEMSET)
      assert omd == d, "OMD must equal the plain dict of its last-value-wins items"
      assert d == omd, "reflected comparison must also be True"
      assert not (omd != d), "__ne__ must agree with __eq__"


  def case_omd_eq_matching_dict():
      omd = OMD([('a', 1), ('b', 2)])
      assert omd == {'a': 1, 'b': 2}, "OMD must equal a dict with identical items"


  def case_omd_eq_todict_roundtrip():
      for itemset in ([], [('a', 1)], [('A', 'One'), ('A', 'One')], _ITEMSET):
          omd = OMD(itemset)
          assert omd == omd.todict(), \
              "OMD must equal its own todict() for itemset %r" % (itemset,)


  def case_omd_ne_dict_missing_key():
      omd = OMD([('a', 1), ('b', 2)])
      assert not (omd == {'a': 1, 'c': 2}), \
          "OMD == dict must be False when a key is missing from the dict"


  def case_omd_ne_dict_different_length():
      omd = OMD([('a', 1), ('b', 2)])
      assert not (omd == {'a': 1}), "OMD == shorter dict must be False"
      assert not (omd == {'a': 1, 'b': 2, 'c': 3}), "OMD == longer dict must be False"


  def case_omd_eq_omd_preserved():
      omd = OMD(_ITEMSET)
      assert omd == omd, "OMD must equal itself"
      assert omd == OMD(_ITEMSET), "equal OMDs must compare equal"
      assert not (omd != OMD(_ITEMSET)), "__ne__ must agree"
      assert omd != OMD(dict(_ITEMSET)), \
          "OMD with duplicate keys must not equal the de-duplicated OMD"
      assert OMD([('a', 1), ('a', 2)]) != OMD([('a', 2), ('a', 1)]), \
          "OMD equality must remain order sensitive"


  def case_omd_ne_non_mapping():
      omd = OMD([('a', 1)])
      assert not (omd == [('a', 1)]), "OMD must not equal a non-mapping"
      assert not (omd == 5), "OMD must not equal a non-sized object"


  def case_empty_omd_eq_empty_dict():
      assert OMD() == {}, "empty OMD must equal an empty dict"
      assert not (OMD() != {}), "__ne__ must agree for empties"


  def case_mapping_raising_keyerror_is_unequal():
      class Mapping:
          def keys(self):
              return ['a']

          def __getitem__(self, k):
              raise KeyError(k)

          def __len__(self):
              return 1

      assert not (OMD([('a', 1)]) == Mapping()), \
          "a mapping whose lookup raises KeyError must compare unequal"


  def case_urlutils_omd_eq_dict_last_value_wins():
      uomd = UrlOMD(_ITEMSET)
      assert uomd == dict(_ITEMSET), \
          "urlutils OMD must still equal its last-value-wins dict"


  # --------------------------------------------------------------------- main

  def _cases():
      return {n: f for n, f in sorted(globals().items())
              if n.startswith('case_') and callable(f)}


  def main(argv):
      cases = _cases()
      if len(argv) > 1:
          name = argv[1]
          if name not in cases:
              print('no such case: %s' % name)
              return 1
          selected = [(name, cases[name])]
      else:
          selected = list(cases.items())

      failed = 0
      for name, func in selected:
          try:
              func()
          except AssertionError as ae:
              failed += 1
              print('FAIL %s: %s' % (name, ae))
          except Exception as e:
              failed += 1
              print('ERROR %s: %s: %s' % (name, type(e).__name__, e))
          else:
              print('ok   %s' % name)
      print('%d/%d passed' % (len(selected) - failed, len(selected)))
      return 1 if failed else 0


  if __name__ == '__main__':
      sys.exit(main(sys.argv))
