  """Probe: tied-key ordering of SortedList/SortedKeyList update vs add."""
  import sys
  sys.path.insert(0, '.')

  from sortedcontainers import SortedList, SortedKeyList


  def mod(val):
      """Key with deliberate ties: 100, 110, 120 ... all have key 0."""
      return val % 10


  class Eq(object):
      """Compares equal on `k`, but stays distinguishable through `tag`."""

      def __init__(self, k, tag):
          self.k = k
          self.tag = tag

      def __lt__(self, other):
          return self.k < other.k

      def __le__(self, other):
          return self.k <= other.k

      def __gt__(self, other):
          return self.k > other.k

      def __ge__(self, other):
          return self.k >= other.k

      def __eq__(self, other):
          return self.k == other.k

      def __ne__(self, other):
          return self.k != other.k

      def __hash__(self):
          return hash(self.k)

      def __repr__(self):
          return self.tag


  class Opaque(object):
      """Carries a key and a tag but supports NO ordering between values.

      Any attempt to order two values raises TypeError, so an implementation
      that lets values take part in a sort -- directly or as a hidden
      tiebreaker inside a decoration -- blows up instead of silently
      reordering.
      """

      __slots__ = ('k', 'tag')

      def __init__(self, k, tag):
          self.k = k
          self.tag = tag

      def _refuse(self, other):
          raise TypeError('Opaque values are not orderable')

      __lt__ = _refuse
      __le__ = _refuse
      __gt__ = _refuse
      __ge__ = _refuse

      def __repr__(self):
          return self.tag


  def tags(seq):
      return [val.tag for val in seq]


  def expected_tie_merge(existing, incoming, key):
      """The one legal result: per key, existing values then incoming values.

      Within a key, existing values keep the relative order they already had
      and incoming values keep the relative order of the input iterable.  That
      is exactly what repeated `add` of the incoming values (in key-sorted
      order) produces; no property of the values themselves may influence it.
      """
      result = []
      keys = [key(val) for val in existing] + [key(val) for val in incoming]
      for cur in sorted(set(keys)):
          result.extend([val for val in existing if key(val) == cur])
          result.extend([val for val in incoming if key(val) == cur])
      return result

  def case_skl_update_tie_after_existing():
      skl = SortedKeyList([10], key=mod)
      skl.update([0])
      assert list(skl) == [10, 0], 'update tie: got %r, want [10, 0]' % (list(skl),)


  def case_skl_update_multiple_ties_after_existing():
      skl = SortedKeyList([100, 110, 120], key=mod)
      skl.update([0, 10])
      want = [100, 110, 120, 0, 10]
      assert list(skl) == want, 'update ties: got %r, want %r' % (list(skl), want)


  def case_skl_update_ties_keep_insertion_order():
      # Existing tied values held in descending value order, and an incoming
      # batch also given in descending value order.  Every value has key 0, so
      # nothing may be reordered: existing keep their positions and incoming
      # keep the iterable's own relative order, appended after them.
      skl = SortedKeyList([30, 20], key=mod)
      skl.update([50, 40])            # big branch: len(values) * 4 >= len(skl)
      want = [30, 20, 50, 40]
      assert list(skl) == want, \
          'update reordered ties: got %r, want %r' % (list(skl), want)

      by_add = SortedKeyList([30, 20], key=mod)
      for val in [50, 40]:
          by_add.add(val)
      assert list(skl) == list(by_add), \
          'update %r disagrees with add %r' % (list(skl), list(by_add))


  def case_skl_update_never_reorders_by_value():

      # each key on both sides, so any value-derived ordering shows up.
      existing = [90, 70, 50, 91, 71, 92]
      incoming = [40, 20, 30, 41, 21, 82, 62]
      skl = SortedKeyList(existing, key=mod)
      skl._reset(4)
      skl.update(incoming)
      want = expected_tie_merge(existing, incoming, mod)
      assert list(skl) == want, \
          'value order leaked: got %r, want %r' % (list(skl), want)
      skl._check()


  def case_skl_update_incoming_order_preserved_per_key():
      # Only the incoming batch has ties; each key's incoming run is descending
      # by value and must survive verbatim.
      existing = [5]
      incoming = [90, 80, 70, 61, 51, 41, 92, 82]
      skl = SortedKeyList(existing, key=mod)
      skl.update(incoming)
      want = expected_tie_merge(existing, incoming, mod)
      assert list(skl) == want, \

Two things closed the hole. R2 replaces the coarse "existing before incoming" claim with the exact per-key concatenation law (existing in existing order, then incoming in input order), checked against a helper that derives the expected sequence rather than a hand-written literal, over tie groups deliberately anti-sorted by value on both sides. R3 adds the structural property the previous spec never stated — a SortedKeyList orders by key alone, so `update` must never compare two values — enforced with an `Opaque` value type that raises `TypeError` on any ordering comparison, which kills the whole decorate-with-value family rather than the one counterexample. The probe also now checks `_reset` with opaque values, since `_reset` routes through `_update`.
