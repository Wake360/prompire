  """Probe cases for appending/inserting into arrays whose commas start each line."""

  import sys

  sys.path.insert(0, ".")

  import tomlkit  # noqa: E402


  def _roundtrip(src, mutate):
      """Parse src, mutate array 'a', dump, reparse -> (out, in_memory, reparsed)."""
      doc = tomlkit.parse(src)
      arr = doc["a"]
      mutate(arr)
      out = tomlkit.dumps(doc)
      try:
          reparsed = tomlkit.parse(out)["a"].unwrap()
      except Exception as e:  # noqa: BLE001
          raise AssertionError(
              f"output is unparseable TOML ({type(e).__name__}: {e}); output={out!r}"
          ) from None
      return out, arr.unwrap(), reparsed


  def _check(src, mutate, expected):
      out, in_mem, reparsed = _roundtrip(src, mutate)
      assert in_mem == expected, f"in-memory value is {in_mem!r}, expected {expected!r}"
      assert reparsed == expected, (
          f"round-trip lost/changed values: reparsed={reparsed!r}, "
          f"expected={expected!r}; output={out!r}"
      )


  # --- behavioral: broken on the current checkout ---------------------------


  def case_append_leading_comma_dangling_comment():
      # Literal report: append yields TOML the parser rejects (double trailing comma).
      _check("a = [\n    1\n    , 2\n    , # c\n]\n", lambda a: a.append(3), [1, 2, 3])


  def case_append_leading_comma_trailing_comment():
      # Appended value lands on the comment line and is silently swallowed.
      _check("a = [\n    1\n    , 2  # two\n]\n", lambda a: a.append(3), [1, 2, 3])


  def case_append_leading_comma_comment_every_line():
      _check(
          "a = [\n    1  # one\n    , 2  # two\n    , 3  # three\n]\n",
          lambda a: a.append(4),
          [1, 2, 3, 4],
      )


  def case_append_leading_comma_string_values():
      _check(
          'a = [\n    "x"  # one\n    , "y"  # two\n]\n',
          lambda a: a.append("z"),
          ["x", "y", "z"],
      )


  def case_append_leading_comma_nested_array():
      _check(
          "a = [\n    [1]\n    , [2]  # two\n]\n",
          lambda a: a.append([3]),
          [[1], [2], [3]],
      )


  def case_extend_leading_comma_trailing_comment():
      _check("a = [\n    1\n    , 2  # two\n]\n", lambda a: a.extend([3, 4]), [1, 2, 3, 4])


  def case_iadd_leading_comma_trailing_comment():
      def mutate(a):
          a += [3]

      _check("a = [\n    1\n    , 2  # two\n]\n", mutate, [1, 2, 3])


  def case_insert_middle_leading_comma_after_comment():
      # The stranded "\n    ," group gets a second comma -> ",," -> unparseable.
      _check("a = [\n    1  # one\n    , 2\n]\n", lambda a: a.insert(1, 99), [1, 99, 2])


  def case_insert_middle_leading_comma_after_own_line_comment():
      _check("a = [\n    1\n    # note\n    , 2\n]\n", lambda a: a.insert(1, 99), [1, 99, 2])


  def case_append_after_inline_value_with_trailing_comment():

      # previous value, so its indent has no newline and the comment eats the append.
      _check("a = [\n    1, 2  # two\n]\n", lambda a: a.append(3), [1, 2, 3])
