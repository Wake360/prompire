```yaml
  """Probe cases for appending/inserting into arrays whose commas start each line."""

  import sys

  sys.path.insert(0, ".")

  import tomlkit  # noqa: E402


  def _roundtrip(src, mutate, label=""):
      """Parse src, mutate array 'a', dump, reparse -> (out, in_memory, reparsed)."""
      where = f" [{label}]" if label else ""
      doc = tomlkit.parse(src)
      arr = doc["a"]
      mutate(arr)
      out = tomlkit.dumps(doc)
      try:
          reparsed = tomlkit.parse(out)["a"].unwrap()
      except Exception as e:  # noqa: BLE001
          raise AssertionError(
              f"output is unparseable TOML{where} ({type(e).__name__}: {e}); "
              f"input={src!r} output={out!r}"
          ) from None
      return out, arr.unwrap(), reparsed


  def _check(src, mutate, expected, label=""):
      where = f" [{label}]" if label else ""
      out, in_mem, reparsed = _roundtrip(src, mutate, label)
      assert in_mem == expected, (
          f"in-memory value is {in_mem!r}, expected {expected!r}{where}"
      )
      assert reparsed == expected, (
          f"round-trip lost/changed values{where}: reparsed={reparsed!r}, "
          f"expected={expected!r}; input={src!r} output={out!r}"
      )
      restable = tomlkit.dumps(tomlkit.parse(out))
      assert restable == out, (
          f"output does not re-serialize byte-identically{where}: "
          f"output={out!r} reserialized={restable!r}"
      )


  def _layouts():
      """Multiline layouts: comma placement x comment style x dangling comma x size.

      Every one of these parses and dumps byte-identically today; the mutation
      behaviour must not depend on which of them is in front of it.
      """
      for lead in (True, False):
          for comment in ("none", "inline", "own_line", "first"):
              for dangling in (False, True):
                  for count in (1, 2, 3):
                      values = list(range(1, count + 1))
                      lines = ["a = ["]
                      for i, v in enumerate(values):
                          last = i == count - 1
                          if lead:
                              line = f"    {v}" if i == 0 else f"    , {v}"
                          else:
                              line = f"    {v}" + ("" if last else ",")
                          if last and comment == "inline":
                              line += "  # c"
                          if i == 0 and not last and comment == "first":
                              line += "  # f"
                          lines.append(line)
                          if last and comment == "own_line":
                              lines.append("    # c")
                      if dangling:
                          lines.append("    ,")
                      lines.append("]")
                      name = (
                          f"lead_comma={lead} comment={comment} "
                          f"dangling={dangling} count={count}"
                      )
                      yield name, "\n".join(lines) + "\n", values

  def case_append_leading_comma_dangling_comment():
      # Literal report: append yields TOML the parser rejects (double trailing comma).
      _check("a = [\n    1\n    , 2\n    , # c\n]\n", lambda a: a.append(3), [1, 2, 3])


  def case_append_leading_comma_trailing_comment():
      # Appended value lands on the comment line and is silently swallowed.
      _check("a = [\n    1\n    , 2  # two\n]\n", lambda a: a.append(3), [1, 2, 3])


  def case_append_leading_comma_comment_every_line():
      _check(
          "a = [\n    1  # one\n    , 2  # two\n    , 3  # three\n]\n",
          lambda a: a.append(4),
          [1, 2, 3, 4],
      )


  def case_append_leading_comma_string_values():
      _check(
          'a = [\n    "x"  # one\n    , "y"  # two\n]\n',
          lambda a: a.append("z"),
          ["x", "y", "z"],
      )


  def case_append_leading_comma_nested_array():

          lambda a: a.append([3]),
          [[1], [2], [3]],
      )


  def case_extend_leading_comma_trailing_comment():

          a += [3]

      # previous value, so its indent has no newline and the comment eats the append.
      _check("a = [\n    1, 2  # two\n]\n", lambda a: a.append(3), [1, 2, 3])


  def case_append_comment_and_dangling_comma_combined():
      # Each ingredient (leading commas, a trailing comment, the dangling comma such
      # arrays idiomatically end with) round-trips on its own; every combination of
      # them must round-trip too. Placing the value on a fresh line is not enough if
      # the comma-bearing indent of the stranded dangling-comma group is reused.
      for src in (
          "a = [\n    1\n    , 2  # two\n    ,\n]\n",
          "a = [\n    1\n    , 2\n    # note\n    ,\n]\n",
          "a = [\n    1,\n    2  # two\n    ,\n]\n",
          "a = [\n    1,\n    2\n    # note\n    ,\n]\n",
      ):
          _check(src, lambda a: a.append(3), [1, 2, 3], src)


  def case_append_after_own_line_comment_keeps_separator():
      # The comment-only group has a Null value, so the separating comma is skipped
      # entirely and two values end up adjacent with no comma between them.
      for src in (
          "a = [\n    1\n    , 2\n    # note\n]\n",
          "a = [\n    1,\n    2\n    # note\n]\n",
      ):
          _check(src, lambda a: a.append(3), [1, 2, 3], src)


  def case_append_every_layout():
      for name, src, values in _layouts():
          _check(src, lambda a: a.append(99), values + [99], name)


  def case_extend_and_iadd_every_layout():
      def iadd(a):
          a += [99]

      for name, src, values in _layouts():
          _check(src, lambda a: a.extend([98, 99]), values + [98, 99], f"extend {name}")
          _check(src, iadd, values + [99], f"+= {name}")


  def case_insert_every_position_every_layout():
      for name, src, values in _layouts():
          for pos in range(len(values) + 1):
              _check(
                  src,
                  lambda a: a.insert(pos, 99),
                  values[:pos] + [99] + values[pos:],
                  f"insert({pos}) {name}",
              )


  def case_repeated_append_every_layout():
      # A single append must not leave the array in a state the next append breaks on.
      def mutate(a):
          a.append(98)
          a.append(99)

      for name, src, values in _layouts():
          _check(src, mutate, values + [98, 99], name)


  def case_append_after_delete_every_layout():
      # Deletion moves commas into the following group's indent; appending after that
      # must still see exactly one separator.
      for name, src, values in _layouts():
          for i in range(len(values)):

              def mutate(a, i=i):
                  del a[i]
                  a.append(99)

              _check(
                  src,
                  mutate,
                  values[:i] + values[i + 1 :] + [99],
                  f"del[{i}] then append {name}",
              )


  def case_append_values_containing_hash_and_comma():
      # '#' and ',' inside a string are data: deciding where the new value goes must
      # come from the parsed structure, not from scanning the rendered text.
      for src, values in (
          ('a = [\n    "a#b"\n    , "c, d"  # c\n]\n', ["a#b", "c, d"]),
          ('a = [\n    "a#b",\n    "c, d"\n    # c\n]\n', ["a#b", "c, d"]),
          ('a = [\n    "a#b"\n    , "c, d"  # c\n    ,\n]\n', ["a#b", "c, d"]),
      ):
          _check(src, lambda a: a.append("e#f"), values + ["e#f"], src)

  def case_no_mutation_roundtrip_every_layout():

      doc["x"].append(2)
      assert doc.as_string() == "x = [\n    1,  # comment\n    2\n]", (
          f"layout changed: {doc.as_string()!r}"
      )


  def case_existing_multiline_flag_append_layout_preserved():
      # mirrors tests/test_items.py::test_append_to_multiline_array_with_comment
      doc = tomlkit.parse("x = [\n    # Here is a comment\n    1,\n    2\n]\n")
      doc["x"].multiline(True).append(3)
      assert (
          doc.as_string() == "x = [\n    # Here is a comment\n    1,\n    2,\n    3,\n]\n"
      ), f"layout changed: {doc.as_string()!r}"


  def main(argv):
      cases = {k: v for k, v in sorted(globals().items()) if k.startswith("case_")}
      names = argv[1:] or list(cases)
      failed = 0
      for name in names:
          fn = cases.get(name)
          if fn is None:
              print(f"FAIL {name}: no such case")
              failed += 1
              continue
          try:
              fn()
          except AssertionError as e:
              failed += 1
              print(f"FAIL {name}: {e}")
          else:
              print(f"pass {name}")
      return 1 if failed else 0


  if __name__ == "__main__":
      sys.exit(main(sys.argv))
```

What changed and why:

- **The hole**: the old spec pinned each ingredient (leading commas, a trailing comment, a dangling comma) separately, so an implementation could special-case "previous group ends in a comment" and "stranded group already has a comma" and still reuse that stranded group's comma-bearing indent. R3 now states the invariant directly — exactly one comma between adjacent values, wherever it is stored, and never re-emit a comma already rendered — and R5 requires layout-independence over a generated matrix of 144 layouts × append/extend/+=/insert-at-every-position/repeated-append/delete-then-append.
- **Adopted the counter-probe** as `case_append_comment_and_dangling_comma_combined`, extended to the two trailing-comma sibling shapes.
- **Found a defect the old probe missed** (R4): appending after an own-line comment omits the separator entirely (`# note` group has a `Null` value, so the comma guard skips it) — this breaks ordinary trailing-comma arrays too, not just leading-comma ones.
- **Added anti-shortcut guards**: R8 (a `#` or `,` inside a string is data, so no rendered-text scanning) and a re-serialization-stability assertion in `_check`.
- **Kept everything that was right**: all 10 original cases, both exact-output regression tests, and the delete/insert-at-0/no-op boundaries — now also swept across every layout, all of which are green today.
- **Out of scope, stated**: `Array.add_line` has an independent pre-existing failure on comment-plus-dangling-comma arrays; the probe only pins its currently-correct behavior.
