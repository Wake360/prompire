  def case_asciidoc_row_bad_arity_raises():
      fmt = _table_formats["asciidoc"]
      for builder in (fmt.datarow, fmt.headerrow):
          for args in ((1,), (1, 2, 3, 4)):
              try:
                  builder(*args)
              except ValueError:
                  continue
              raise AssertionError(f"expected ValueError for asciidoc row builder{args!r}")


  CASES = {k: v for k, v in sorted(globals().items()) if k.startswith("case_")}


  def _run(name):
      try:
          CASES[name]()
      except AssertionError as e:
          print(f"FAIL {name}: {e}")
          return False
      print(f"ok   {name}")
      return True


  if __name__ == "__main__":
      names = sys.argv[1:] or list(CASES)
      unknown = [n for n in names if n not in CASES]
      if unknown:
          print(f"unknown case(s): {unknown}")
          sys.exit(1)
      sys.exit(0 if all([_run(n) for n in names]) else 1)
