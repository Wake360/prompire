"""Counter-probe: update must not reorder tied values by value comparison."""
import sys
sys.path.insert(0, '.')

from sortedcontainers import SortedKeyList


def mod(val):
    """Key with deliberate ties: 20, 30, 40, 50 all have key 0."""
    return val % 10


def case_skl_update_ties_keep_insertion_order():
    # Existing tied values held in descending value order, and an incoming
    # batch also given in descending value order.  Every value has key 0, so
    # nothing may be reordered: existing keep their positions and incoming
    # keep the iterable's own relative order, appended after them.
    skl = SortedKeyList([30, 20], key=mod)
    skl.update([50, 40])            # big branch: len(values) * 4 >= len(skl)
    want = [30, 20, 50, 40]
    assert list(skl) == want, \
        'update reordered ties: got %r, want %r' % (list(skl), want)

    by_add = SortedKeyList([30, 20], key=mod)
    for val in [50, 40]:
        by_add.add(val)
    assert list(skl) == list(by_add), \
        'update %r disagrees with add %r' % (list(skl), list(by_add))


CASES = dict((name, obj) for name, obj in sorted(globals().items())
             if name.startswith('case_') and callable(obj))


def main(argv):
    names = argv[1:] if len(argv) > 1 else sorted(CASES)
    failed = 0
    for name in names:
        if name not in CASES:
            print('UNKNOWN %s' % name)
            failed += 1
            continue
        try:
            CASES[name]()
        except AssertionError as exc:
            print('FAIL %s: %s' % (name, exc))
            failed += 1
        else:
            print('PASS %s' % name)
    return 1 if failed else 0


if __name__ == '__main__':
    sys.exit(main(sys.argv))
