  """Probe: tied-key ordering of SortedList/SortedKeyList update vs add."""
  import sys
  sys.path.insert(0, '.')

  from sortedcontainers import SortedList, SortedKeyList


  def mod(val):
      """Key with deliberate ties: 100, 110, 120 ... all have key 0."""
      return val % 10


  class Eq(object):
      """Compares equal on `k`, but stays distinguishable through `tag`."""

      def __init__(self, k, tag):
          self.k = k
          self.tag = tag

      def __lt__(self, other):
          return self.k < other.k

      def __le__(self, other):
          return self.k <= other.k

      def __gt__(self, other):
          return self.k > other.k

      def __ge__(self, other):
          return self.k >= other.k

      def __eq__(self, other):
          return self.k == other.k

      def __ne__(self, other):
          return self.k != other.k

      def __hash__(self):
          return hash(self.k)

      def __repr__(self):
          return self.tag


  def tags(seq):
      return [v.tag for v in seq]


  # --- behavioral: update must place tied values after the existing ones ------

  def case_skl_update_tie_after_existing():
      skl = SortedKeyList([10], key=mod)
      skl.update([0])
      assert list(skl) == [10, 0], 'update tie: got %r, want [10, 0]' % (list(skl),)


  def case_skl_update_multiple_ties_after_existing():
      skl = SortedKeyList([100, 110, 120], key=mod)
      skl.update([0, 10])
      want = [100, 110, 120, 0, 10]
      assert list(skl) == want, 'update ties: got %r, want %r' % (list(skl), want)


  def case_skl_update_matches_add():
      incoming = [0, 10, 20, 1, 11]
      by_update = SortedKeyList([100, 110, 101], key=mod)
      by_update.update(incoming)
      by_add = SortedKeyList([100, 110, 101], key=mod)
      for val in sorted(incoming, key=mod):
          by_add.add(val)
      assert list(by_update) == list(by_add), \
          'update %r disagrees with add %r' % (list(by_update), list(by_add))


  def case_skl_update_big_branch_at_scale():
      existing = [10 * i for i in range(1, 51)]
      incoming = [10 * i for i in range(51, 101)]
      skl = SortedKeyList(existing, key=mod)
      skl._reset(4)
      skl.update(incoming)
      want = existing + incoming
      assert list(skl) == want, 'scaled update: got %r, want %r' % (list(skl), want)
      skl._check()


  def case_skl_iadd_tie_after_existing():
      skl = SortedKeyList([100], key=mod)
      skl += [0]
      assert list(skl) == [100, 0], 'iadd tie: got %r, want [100, 0]' % (list(skl),)


  def case_sl_update_tie_after_existing():
      sl = SortedList([Eq(1, 'old')])
      sl.update([Eq(1, 'new')])
      assert tags(sl) == ['old', 'new'], \
          'SortedList update tie: got %r, want [old, new]' % (tags(sl),)


  def case_sl_update_matches_add():
      def build():
          return [Eq(1, 'old0'), Eq(1, 'old1'), Eq(2, 'old2')]

      incoming = [('new0', 1), ('new1', 1), ('new2', 2)]
      by_update = SortedList(build())
      by_update.update([Eq(k, t) for t, k in incoming])
      by_add = SortedList(build())
      for tag, k in incoming:
          by_add.add(Eq(k, tag))
      assert tags(by_update) == tags(by_add), \
          'SortedList update %r disagrees with add %r' % (tags(by_update), tags(by_add))


  # --- regression / boundary: already correct today, must stay correct --------

  def case_skl_add_puts_tie_after_existing():
      skl = SortedKeyList([100, 110], key=mod)
      skl.add(0)
      assert list(skl) == [100, 110, 0], \
          'add tie: got %r, want [100, 110, 0]' % (list(skl),)


  def case_skl_update_on_empty_preserves_input_order():
      skl = SortedKeyList(key=mod)
      skl.update([10, 0, 20])
      assert list(skl) == [10, 0, 20], \
          'empty-receiver update: got %r, want [10, 0, 20]' % (list(skl),)


  def case_skl_init_preserves_input_order():
      skl = SortedKeyList([30, 10, 20], key=mod)
      assert list(skl) == [30, 10, 20], \
          'init order: got %r, want [30, 10, 20]' % (list(skl),)


  def case_skl_update_small_branch_unchanged():
      existing = [10 * i for i in range(1, 101)]
      skl = SortedKeyList(existing, key=mod)
      skl.update([0])
      want = existing + [0]
      assert list(skl) == want, 'small-branch update: got %r, want %r' % (list(skl), want)


  def case_skl_reset_preserves_order():
      skl = SortedKeyList([100, 110, 120, 5], key=mod)
      before = list(skl)
      skl._reset(4)
      assert list(skl) == before, 'reset reordered: %r -> %r' % (before, list(skl))
      skl._check()


  def case_skl_update_keeps_invariants():
      skl = SortedKeyList([100, 110, 3], key=mod)
      skl._reset(4)
      skl.update([0, 10, 13, 7])
      skl._check()
      assert len(skl) == 7, 'len: got %r, want 7' % (len(skl),)
      keys = [mod(v) for v in skl]
      assert keys == sorted(keys), 'keys not ordered: %r' % (keys,)


  def case_skl_add_order_stable_for_sorted_input():
      skl = SortedKeyList(key=mod)
      for val in [100, 110, 120]:
          skl.add(val)
      assert list(skl) == [100, 110, 120], \
          'add insertion order: got %r' % (list(skl),)


  CASES = dict((name, obj) for name, obj in sorted(globals().items())
               if name.startswith('case_') and callable(obj))


  def main(argv):
      names = argv[1:] if len(argv) > 1 else sorted(CASES)
      failed = 0
      for name in names:
          if name not in CASES:
              print('UNKNOWN %s' % name)
              failed += 1
              continue

              CASES[name]()
          except AssertionError as exc:
              print('FAIL %s: %s' % (name, exc))
              failed += 1
          else:
              print('PASS %s' % name)
      return 1 if failed else 0


  if __name__ == '__main__':
      sys.exit(main(sys.argv))
